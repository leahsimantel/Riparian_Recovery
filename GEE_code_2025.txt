// Cloud/Shadow/Snow mask for Landsat surface reflectance
// ============================================================
 
// Single point of control for NDSI (snow) threshold:
var NDSI_THRESHOLD = 0.12 // EDIT NDSI (SNOW) THRESHOLD HERE //  *In Gorge Ck, NDSI ranged [-0.3 - 0.15]. Mid: [-.47 to -.01]

// Define function for cloud, shadow, and snow masking Landsat imagery:
var cloudMask = function(image) {
  var qa = image.select('QA_PIXEL').toInt();  // Landsat Collection 2 SR bit flags

  var cloudShadow = qa.bitwiseAnd(1 << 3).neq(0);
  var snow        = qa.bitwiseAnd(1 << 4).neq(0);  // snow option 1 uses QA pixel.
    // Trying a second approach for snow-masking: NDSI
    // a) Add NDSI band: (Green - SWIR1) / (Green + SWIR1)
  var addNDSI = function(image) {
    var spacecraft = ee.String(image.get('SPACECRAFT_ID'));

    // Dynamically select band names based on sensor
    var greenBand = ee.Algorithms.If(
      spacecraft.equals('LANDSAT_8'), 'SR_B3', 'SR_B2'
    );
    var swir1Band = ee.Algorithms.If(
      spacecraft.equals('LANDSAT_8'), 'SR_B6', 'SR_B5'
    );

    // Compute NDSI with selected bands
    var ndsi = image.normalizedDifference([greenBand, swir1Band]).rename('NDSI');
    return image.addBands(ndsi);
  };
    // b) Define a snow mask based on NDSI threshold (edit TH above start of function).
  var refinedSnowMask = function(image) {
    image = addNDSI(image);
    var ndsi = image.select('NDSI');
    return ndsi.lt(NDSI_THRESHOLD);  // Dynamic threshold
  };
  
  var cloud       = qa.bitwiseAnd(1 << 5).neq(0);

  // Only mask shadow if near clouds (within 5 pixels)
  var cloudBuffer = cloud.focal_max({radius: 5, units: 'pixels'});
  var shadowNearCloud = cloudShadow.and(cloudBuffer);

  // Final clear mask: not cloud, not snow, not shadow near cloud
  //var clear = cloud.not().and(snow.not()).and(shadowNearCloud.not());  // snow option 1 (QA pixel)
  var clear = cloud.not().and(refinedSnowMask(image)).and(shadowNearCloud.not());  // snow option 2 (NDSI)

  // Note on snow masking: QA pixel approach was way too aggressive, even in August imagery. 
  // Will be opting for NDSI with conservative threshold.

  var optical = image.select('SR_B.').multiply(0.0000275).add(-0.2)  // Scaling for SR
                    .updateMask(clear);

  return image.addBands(optical, null, true);
};

// Define water mask application function

// var applyWaterMask = function(image) {
//   return image.updateMask(landMask);
// };

var applyWaterMask = function(image) {
  var proj = image.select(0).projection();   // the Landsat scene’s native UTM 30m grid
  // Align the static land mask to the image grid; no resample() needed
  var landMaskAligned = landMask
      .unmask(1)                 // treat missing JRC as land so we don't over-mask
      .reproject({crs: proj});   // snap to the scene grid

  return image.updateMask(landMaskAligned);
};

// Band renaming for Landsat 8

var bandRenamel8 = function(image) {
  return image.select(['SR_B2', 'SR_B3', 'SR_B4', 'SR_B5', 'SR_B6', 'SR_B7'],
                      ['SR_B1', 'SR_B2', 'SR_B3', 'SR_B4', 'SR_B5', 'SR_B7']);
};

// Function to add HUC12 name to each pixel
var addHUC12Name = function(feature) {
  var huc12Feature = huc12.filterBounds(feature.geometry()).first();
  var huc12Name = ee.Algorithms.If(
    huc12Feature, 
    ee.String(huc12Feature.get('name')), 
    'None'
  );
  return feature.set('huc12', huc12Name);
};

// Function to add Fire Name, Year, and Size (ha) to each pixel
var addFireInfo = function(feature) {
  var fireFeature = fire_atlas.filterBounds(feature.geometry()).first();
  
  var fireName = ee.Algorithms.If(
    fireFeature, 
    ee.String(fireFeature.get('FireNam')), 
    'None'
  );

  var fireYear = ee.Algorithms.If(
    fireFeature, 
    ee.Number(fireFeature.get('Fire_Year')), 
    0
  );

  var fireSizeHa = ee.Algorithms.If(
    fireFeature, 
    ee.Number(fireFeature.get('Size_ha')), 
    null
  );
 
  return feature.set({
    'fire_name': fireName,
    'fire_year': fireYear,
    'fire_size_ha': fireSizeHa
  });
};

// NDVI calculation function
var ndviCalc = function(x) {
  return x.normalizedDifference(['SR_B4', 'SR_B3'])
          .rename('NDVI')
          .set('system:time_start', x.get('system:time_start'));
};

  region: streams_wide.geometry(), // Use the stream geometry as the region of interest
  scale: 30,                       // Landsat resolution
  projection: annualStack.projection(), // Retain the original projection
  geometries: true                 // Keep pixel geometries (lat/lon)
}).map(function(feature) {
  
  // Extract geometry for latitude and longitude
  var coords = feature.geometry().coordinates();
  var lat = ee.Number(coords.get(1)); // Extract latitude
  var lon = ee.Number(coords.get(0)); // Extract longitude
  
  // Generate a unique pixel ID using lat/lon
  var latStr = lat.format('%.7f'); // Format latitude to 7 decimal places
  var lonStr = lon.format('%.7f'); // Format longitude to 7 decimal places
  var pixelID = ee.String(latStr).cat('_').cat(lonStr); // Combine lat/lon for unique ID
  
  // Add pixel ID, latitude, and longitude to the feature
  return feature.set({
    'latitude': lat,   // Add latitude as a property
    'longitude': lon,  // Add longitude as a property
    'pixel_ID': pixelID // Add unique pixel ID as a property
  });
});

//  ===================================================

// Define range of years for climatology
var climYears = ee.List.sequence(1986, 2015);
var months = ee.List.sequence(4, 8);  // April to August

// Function to get monthly mean CWD image (across years) for a given month
var makeMonthlyCWDMean = function(month) {
  month = ee.Number(month);
  var paddedMonth = month.format('%02d');

  var monthlyImages = climYears.map(function(year) {
    year = ee.Number(year);
    return ee.ImageCollection('IDAHO_EPSCOR/TERRACLIMATE')
      .filter(ee.Filter.calendarRange(year, year, 'year'))
      .filter(ee.Filter.calendarRange(month, month, 'month'))
      .select('def')
      .first();  // Grab the first image for that month/year
  });

  // Compute mean image and rename band
  var meanImg = ee.ImageCollection.fromImages(monthlyImages)
    .mean()
    .reproject({crs: refProjection.crs(), scale: 30})
    .rename(ee.String('cwd_30yrAvg_TC_').cat(paddedMonth));

  return meanImg;
};

// Create a list of 5 mean images (one per month)
var monthlyCWDMeans = ee.ImageCollection.fromImages(
  months.map(makeMonthlyCWDMean)
);

// Stack them into a multi-band image
var cwdMonthlyMeansStack = monthlyCWDMeans.toBands();

// Manually remove the index prefix (e.g., '0_', '1_') from band names
cwdMonthlyMeansStack = cwdMonthlyMeansStack.rename(
  cwdMonthlyMeansStack.bandNames().map(function(name) {
    // Remove everything up to and including the first underscore
    return ee.String(name).split('_').slice(1).join('_');
  })
);

// Visual check:

var cwdMonthlyMeans_resampled = cwdMonthlyMeansStack.resample('bilinear');

// RENAME BANDS:
// Function to strip the old prefix and rebuild as <var>_YYYYMM_PR
function renameBands(image, varName) {
  var oldNames = image.bandNames();  // e.g., ["ppt_198401", "ppt_198402", ...]
  var newNames = oldNames.map(function(k) {
    var yyyymm = ee.String(k).replace(varName + '_', '');
    return ee.String(varName).cat('_').cat(yyyymm).cat('_PR');
  });
  return image.rename(newNames);
}

var sampledWithAttributes = sampledPixels.map(function(feature) {
  var point = feature.geometry();
  
  // sample a single pixel; buffer slightly to avoid vertex/edge weirdness
  var sampleGeom = point.buffer(10); // pixel edge buffer, in m
  
  // Fire Severity (numeric)
  var severityValue = severity30.reduceRegion({
    reducer: ee.Reducer.mean(),   
    geometry: sampleGeom,
    scale: 30,
    maxPixels: 1e13
  }).get('sev_num'); 
  
  // Extract severity value from severity_Cannon layer
  var severityCannonValue = sevCannon30.reduceRegion({
    reducer: ee.Reducer.mean(),
    geometry: sampleGeom,
    scale: 30,
    maxPixels: 1e13
  }).get('sev_cannon_num');

  // Assign severity class based on Parks et al. classification:
  var sevClass = ee.Algorithms.If(
      ee.Algorithms.IsEqual(severityValue, null), 'None',  // Class null sev values as 'None'
      ee.Algorithms.If(
          ee.Number(severityValue).lt(0.1), 'Unburned/Very Low',  // CBI 0.0 – 0.1
          ee.Algorithms.If(
              ee.Number(severityValue).lt(1.25), 'Low',           // CBI 0.1 – 1.25
              ee.Algorithms.If(
                  ee.Number(severityValue).lt(2.25), 'Moderate',  // CBI 1.25 – 2.25
                  'High'                                          // CBI 2.25 – 3.0+
              )
          )
      )
  );

  var twiValue = twiResampled.reduceRegion({
    reducer: ee.Reducer.mean(),
    geometry: sampleGeom,
    scale: 30,
    crs: refProjection.crs(),
    bestEffort: true
  }).get('twi');

  var hliValue = hliResampled.reduceRegion({
    reducer: ee.Reducer.mean(),
    geometry: sampleGeom,
    scale: 30,
    crs: refProjection.crs(),
    bestEffort: true
  }).get('hli');

  var cwdValue = cwdResampled.reduceRegion({  // TopoTerra dataset
    reducer: ee.Reducer.mean(),
    geometry: feature.geometry(),
    scale: 30,
    crs: refProjection.crs(),
    bestEffort: true
  }).get('cwd_30yrAvg_TT');
  
  var numBurnValue = num_times_burned.reduceRegion({
    reducer: ee.Reducer.mean(),
    geometry: feature.geometry(),
    scale: 30,
    crs: refProjection.crs(),
    bestEffort: true
  }).get('b1');  

  // Extract nearest stream feature based on the pixel location
  var nearestStream = streams_wide.filterBounds(feature.geometry()).first();
  var fcode = ee.Algorithms.If(nearestStream, nearestStream.get('FCODE'), null);
  var comid = ee.Algorithms.If(nearestStream, nearestStream.get('COMID'), null);

  // -------------------------------
  // Note: TerraClimate monthly values were already added in earlier code
  // -------------------------------
  
  // -------------------------------
  // PRISM monthly values
  // -------------------------------
  
  var pptValues = prism_ppt.reduceRegion({
    reducer: ee.Reducer.first(),
    geometry: point,
    scale: 30,
    crs: refProjection.crs(),
    maxPixels: 1e13
  });

  var tmaxValues = prism_tmax.reduceRegion({
    reducer: ee.Reducer.first(),
    geometry: point,
    scale: 30,
    crs: refProjection.crs(),
    maxPixels: 1e13
  });

  var tmeanValues = prism_tmean.reduceRegion({
    reducer: ee.Reducer.first(),
    geometry: point,
    scale: 30,
    crs: refProjection.crs(),
    maxPixels: 1e13
  });

  // Add to feature 
  return addFireInfo(
    addHUC12Name(
      feature
        .set('sev_num', severityValue)
        .set('sev_cannon_num', severityCannonValue)
        .set('sev_class', sevClass)
        .set('twi', twiValue)
        .set('hli', hliValue)
        .set('cwd_30yrAvg_TT', cwdValue)   
        .set('num_of_burns', numBurnValue)
        .set('FCODE', fcode)
        .set('COMID', comid)
        // PRISM monthlies 
        .set(pptValues)
        .set(tmaxValues)
        .set(tmeanValues)
    )
  );
});
